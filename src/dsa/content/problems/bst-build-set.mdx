Build your own binary search tree from scratch: a generic set that keeps its elements in a BST of nodes you define.

`BSTSetBuild.SearchTreeSet<Element: Comparable>`, a struct with:

- `init()`, `count`, `isEmpty` and `contains(_:)`.
- `insert(_:) -> Bool`: adds the element and returns `true`, or returns `false` if it was already there.
- `remove(_:) -> Element?`: removes the element and returns it, or returns `nil` if it wasn't there.
- `min` and `max`: the smallest and largest elements, or `nil` when empty.
- `floor(_:)`: the largest element less than or equal to the argument, or `nil`.
- `ceiling(_:)`: the smallest element greater than or equal to the argument, or `nil`.
- `inOrder: [Element]`: every element, in increasing order.

`count` and `isEmpty` should be O(1), `inOrder` O(n), and everything else O(h), where h is the height. Copy-on-write isn't required: the tests never copy a set and then change both copies.

## Examples

| Code | Result |
| --- | --- |
| `var set = SearchTreeSet<Int>()` then insert 50, 30, 70, 20, 40 | `set.inOrder == [20, 30, 40, 50, 70]` |
| `set.insert(30)` | `false`, and `count` stays 5 |
| `set.floor(45)`, `set.ceiling(45)` | `Optional(40)`, `Optional(50)` |
| `set.floor(10)` | `nil` |
| `set.remove(30)` | `Optional(30)`; `inOrder` is now `[20, 40, 50, 70]` |
| `set.remove(30)` | `nil` |
| `set.min`, `set.max` | `Optional(20)`, `Optional(70)` |

It works for any `Comparable` element, so a `SearchTreeSet<String>` keeps words in alphabetical order.

## Constraints

- Up to 100,000 operations, with up to 50,000 elements at once.
- Tests insert elements in sorted order only up to 1,000 at a time, so recursion is fine, but loops are better practice: a degenerate tree is as deep as it is long.

<Hints>
<Hint>
Nest a `final class Node` inside the struct, with `value: Element` and optional `left` and `right` children, and give the struct a `root: Node?` and a stored `count`. A class lets several references (a parent, a loop variable) reach the same node.
</Hint>
<Hint>
`contains`, `insert`, `min`, `max`, `floor` and `ceiling` are all one walk down from the root. For `floor`, every node smaller than the argument is a candidate: record it and go right to look for a bigger one; a bigger node sends you left.
</Hint>
<Hint>
For `remove`, walk down remembering the parent. A node with at most one child is replaced by that child (or `nil`) in its parent, or in `root`. A node with two children takes its successor's value, the leftmost node of its right subtree, and the successor (which has no left child) is unlinked instead.
</Hint>
</Hints>

## Solutions

<Approach title="Class nodes in a struct, with loops" time="O(h) per operation" space="O(n)">

Every operation walks one path, with loops so a degenerate tree can't overflow the stack. Comparisons use only `<` and `==`, which `Comparable` provides. DSAKit's `BinarySearchTree` is the finished version, with copy-on-write, `Sequence` and more.

```swift
public enum BSTSetBuild {
    public struct SearchTreeSet<Element: Comparable> {
        final class Node {
            var value: Element
            var left: Node?
            var right: Node?

            init(_ value: Element) {
                self.value = value
            }
        }

        private var root: Node?
        public private(set) var count = 0

        public init() {}

        public var isEmpty: Bool {
            root == nil
        }

        public func contains(_ element: Element) -> Bool {
            var current = root
            while let node = current {
                if element < node.value {
                    current = node.left
                } else if node.value < element {
                    current = node.right
                } else {
                    return true
                }
            }
            return false
        }

        @discardableResult
        public mutating func insert(_ element: Element) -> Bool {
            guard var node = root else {
                root = Node(element)
                count = 1
                return true
            }
            while true {
                if element < node.value {
                    guard let left = node.left else {
                        node.left = Node(element)
                        break
                    }
                    node = left
                } else if node.value < element {
                    guard let right = node.right else {
                        node.right = Node(element)
                        break
                    }
                    node = right
                } else {
                    return false
                }
            }
            count += 1
            return true
        }

        @discardableResult
        public mutating func remove(_ element: Element) -> Element? {
            var parent: Node? = nil
            var current = root
            while let node = current, node.value != element {
                parent = node
                current = element < node.value ? node.left : node.right
            }
            guard let node = current else { return nil }
            let removed = node.value
            if let right = node.right, node.left != nil {
                var successorParent = node
                var successor = right
                while let smaller = successor.left {
                    successorParent = successor
                    successor = smaller
                }
                node.value = successor.value
                if successorParent === node {
                    node.right = successor.right
                } else {
                    successorParent.left = successor.right
                }
            } else {
                let child = node.left ?? node.right
                if let parent {
                    if parent.left === node {
                        parent.left = child
                    } else {
                        parent.right = child
                    }
                } else {
                    root = child
                }
            }
            count -= 1
            return removed
        }

        public var min: Element? {
            guard var node = root else { return nil }
            while let left = node.left {
                node = left
            }
            return node.value
        }

        public var max: Element? {
            guard var node = root else { return nil }
            while let right = node.right {
                node = right
            }
            return node.value
        }

        public func floor(_ element: Element) -> Element? {
            var best: Element? = nil
            var current = root
            while let node = current {
                if node.value == element {
                    return node.value
                } else if node.value < element {
                    best = node.value
                    current = node.right
                } else {
                    current = node.left
                }
            }
            return best
        }

        public func ceiling(_ element: Element) -> Element? {
            var best: Element? = nil
            var current = root
            while let node = current {
                if node.value == element {
                    return node.value
                } else if node.value > element {
                    best = node.value
                    current = node.left
                } else {
                    current = node.right
                }
            }
            return best
        }

        public var inOrder: [Element] {
            var result: [Element] = []
            var stack: [Node] = []
            var current = root
            while current != nil || !stack.isEmpty {
                while let node = current {
                    stack.append(node)
                    current = node.left
                }
                let node = stack.removeLast()
                result.append(node.value)
                current = node.right
            }
            return result
        }
    }
}
```

</Approach>
