Write a generic singly linked list class that keeps a head, a tail and a count, so that adding at either end and asking for the count or the last element are all O(1).

`LinkedListFromScratch.List<Element>`, with `count`, `isEmpty`, `first`, `last`, `append(_:)`, `prepend(_:)`, `insert(_:at:)`, `removeFirst()`, `remove(at:)`, `value(at:)`, `reverse()`, `makeIterator()`, and, when `Element` is `Equatable`, `firstIndex(of:)` and `removeAll(equalTo:)`

The starter code gives you the `Node` class, the `head` and `tail` properties, the initialisers and an `Iterator` struct to finish. Implement the rest:

- Positions count from 0. `insert(_:at:)` accepts `0...count` (where `count` means "at the end") and returns `true`; any other position returns `false` and changes nothing.
- `remove(at:)`, `removeFirst()` and `value(at:)` return `nil` for a position outside `0..<count` or an empty list.
- `reverse()` turns the list around in place, without creating nodes.
- `removeAll(equalTo:)` removes every matching element and returns how many it removed.
- The iterator walks from head to tail, so `for value in list` and `Array(list)` work.

`count` starts as a computed property returning 0: turn it into a stored property that every method keeps up to date. `reverse()` and `removeAll(equalTo:)` use ideas from [Reverse a list, iteratively and recursively](/dsa/learn/ll-reverse) and [Dummy and sentinel nodes](/dsa/learn/ll-sentinel); leave them for later if you haven't reached those lessons.

## Examples

| Code | Result |
| --- | --- |
| `let list = List([1, 2, 3])`, `list.insert(9, at: 1)` | `true`; `Array(list)` is `[1, 9, 2, 3]` |
| `list.insert(5, at: 7)` | `false`; the list is unchanged |
| `list.remove(at: 3)` | `3`; `list.last` is now `2` |
| then `list.append(4)` | `Array(list)` is `[1, 9, 2, 4]` |
| `List([7, 1, 7, 7]).removeAll(equalTo: 7)` | `3`, leaving `[1]` |
| `List(["a", "b"])`, then `reverse()` | `first` is `"b"`, `last` is `"a"` |
| `List<Int>().removeFirst()` | `nil` |

## Constraints

- Up to 100,000 elements; `append`, `prepend`, `count`, `first` and `last` must be O(1).
- Don't store the elements in an array: use the nodes.

## Solutions

<Approach title="Head, tail and count" time="O(1) at the ends, O(i) at position i" space="O(n)">

The tail makes `append` O(1), the stored count makes `count` O(1), and a private helper walks to a position for the middle operations. Each method only has to remember which of `head`, `tail` and `count` it might change.

```swift
public enum LinkedListFromScratch {
    public final class List<Element>: Sequence {
        final class Node {
            var value: Element
            var next: Node?

            init(_ value: Element, next: Node? = nil) {
                self.value = value
                self.next = next
            }
        }

        var head: Node?
        var tail: Node?
        public private(set) var count = 0

        public init() {}

        public convenience init(_ elements: [Element]) {
            self.init()
            for element in elements {
                append(element)
            }
        }

        public var isEmpty: Bool { head == nil }
        public var first: Element? { head?.value }
        public var last: Element? { tail?.value }

        public func append(_ value: Element) {
            let node = Node(value)
            if let tail {
                tail.next = node
            } else {
                head = node
            }
            tail = node
            count += 1
        }

        public func prepend(_ value: Element) {
            head = Node(value, next: head)
            if tail == nil {
                tail = head
            }
            count += 1
        }

        private func node(at index: Int) -> Node {
            var node = head!
            for _ in 0..<index {
                node = node.next!
            }
            return node
        }

        @discardableResult
        public func insert(_ value: Element, at index: Int) -> Bool {
            guard index >= 0 && index <= count else { return false }
            if index == 0 {
                prepend(value)
            } else if index == count {
                append(value)
            } else {
                let before = node(at: index - 1)
                before.next = Node(value, next: before.next)
                count += 1
            }
            return true
        }

        @discardableResult
        public func removeFirst() -> Element? {
            guard let first = head else { return nil }
            head = first.next
            if head == nil {
                tail = nil
            }
            count -= 1
            return first.value
        }

        @discardableResult
        public func remove(at index: Int) -> Element? {
            guard index >= 0 && index < count else { return nil }
            if index == 0 {
                return removeFirst()
            }
            let before = node(at: index - 1)
            let removed = before.next!
            before.next = removed.next
            if removed === tail {
                tail = before
            }
            count -= 1
            return removed.value
        }

        public func value(at index: Int) -> Element? {
            guard index >= 0 && index < count else { return nil }
            return node(at: index).value
        }

        public func reverse() {
            var previous: Node? = nil
            var current = head
            tail = head
            while let node = current {
                let following = node.next
                node.next = previous
                previous = node
                current = following
            }
            head = previous
        }

        public struct Iterator: IteratorProtocol {
            var current: Node?

            public mutating func next() -> Element? {
                guard let node = current else { return nil }
                current = node.next
                return node.value
            }
        }

        public func makeIterator() -> Iterator {
            Iterator(current: head)
        }
    }
}

extension LinkedListFromScratch.List where Element: Equatable {
    public func firstIndex(of value: Element) -> Int? {
        var index = 0
        var current = head
        while let node = current {
            if node.value == value {
                return index
            }
            index += 1
            current = node.next
        }
        return nil
    }

    @discardableResult
    public func removeAll(equalTo value: Element) -> Int {
        guard let first = head else { return 0 }
        let dummy = Node(first.value, next: head)
        var previous = dummy
        var removed = 0
        while let node = previous.next {
            if node.value == value {
                previous.next = node.next
                removed += 1
            } else {
                previous = node
            }
        }
        head = dummy.next
        tail = head == nil ? nil : previous
        count -= removed
        return removed
    }
}
```

In `removeAll(equalTo:)`, `previous` ends on the last node that was kept, which is exactly the new tail, unless nothing was kept at all.

</Approach>
